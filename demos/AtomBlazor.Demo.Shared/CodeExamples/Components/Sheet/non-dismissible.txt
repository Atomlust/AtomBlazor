    <SheetTrigger>
        <Button Variant="ButtonVariant.Outline">Open Sheet</Button>
    </SheetTrigger>
    <SheetContent>
        <SheetHeader>
            <SheetTitle>Non-Dismissible</SheetTitle>
            <SheetDescription>
                Clicking the backdrop or pressing Escape will not close this sheet.
            </SheetDescription>
        </SheetHeader>
        <SheetFooter>
            <SheetClose>
                <Button>Got it</Button>
            </SheetClose>
        </SheetFooter>
    </SheetContent>
</Sheet>
