@* products = MockDataService.GenerateProducts(60)
   Both ListTemplate and GridTemplate are provided, so the toolbar
   shows layout-toggle buttons letting the user switch between views. *@
<DataView TItem="Product" Data="@products" InitialPageSize="12"
            PageSizes="@(new[] { 6, 12, 24, 48 })">
    <ListTemplate Context="product">
        <Card Class="flex items-center gap-4 p-3 shadow-none hover:bg-accent/50 transition-colors">
            <div class="size-16 shrink-0 rounded-md overflow-hidden bg-muted">
                <img src="@product.ImageUrl" alt="@product.Name"
                     class="object-cover w-full h-full" loading="lazy" />
            </div>
            <div class="flex-1 min-w-0">
                <div class="flex items-center gap-2">
                    <p class="text-sm font-semibold truncate">@product.Name</p>
                    <Badge Variant="BadgeVariant.Outline" Class="shrink-0 text-xs">@product.Category</Badge>
                </div>
                <Rating Value="@product.Rating" ReadOnly="true" Size="RatingSize.Small" />
            </div>
            <div class="flex flex-col items-end gap-2 shrink-0">
                <span class="text-base font-bold">$@product.Price.ToString("F2")</span>
                @if (product.Stock > 0)
                {
                    <Button Size="ButtonSize.Small">Add to Cart</Button>
                }
                else
                {
                    <Badge Variant="BadgeVariant.Destructive">Out of Stock</Badge>
                }
            </div>
        </Card>
    </ListTemplate>
    <GridTemplate Context="product">
        <Card Class="overflow-hidden flex flex-col hover:shadow-md transition-shadow">
            <div class="relative aspect-square overflow-hidden bg-muted">
                <img src="@product.ImageUrl" alt="@product.Name"
                     class="object-cover w-full h-full" loading="lazy" />
                @if (product.Stock == 0)
                {
                    <div class="absolute inset-0 bg-background/70 flex items-center justify-center">
                        <TypographyMuted>Out of Stock</TypographyMuted>
                    </div>
                }
            </div>
            <CardHeader>
                <div class="flex items-start justify-between gap-2">
                    <CardTitle Class="text-base line-clamp-2 leading-snug">@product.Name</CardTitle>
                    <Badge Variant="BadgeVariant.Outline" Class="shrink-0 text-xs mt-0.5">@product.Category</Badge>
                </div>
                <CardDescription Class="line-clamp-2">@product.Description</CardDescription>
            </CardHeader>
            <CardContent>
                <Rating Value="@product.Rating" ReadOnly="true" Size="RatingSize.Small" />
            </CardContent>
            <CardFooter Class="justify-between">
                <span class="text-base font-bold">$@product.Price.ToString("F2")</span>
                <Button Size="ButtonSize.Small"
                          Variant="@(product.Stock > 0 ? ButtonVariant.Default : ButtonVariant.Outline)"
                          Disabled="@(product.Stock == 0)">
                    @(product.Stock > 0 ? "Add to Cart" : "Notify Me")
                </Button>
            </CardFooter>
        </Card>
    </GridTemplate>
    <Fields>
        <DataViewColumn TItem="Product" TValue="string"  Property="@(p => p.Name)"     Header="Name"     Sortable Filterable />
        <DataViewColumn TItem="Product" TValue="string"  Property="@(p => p.Category)" Header="Category" Sortable Filterable />
        <DataViewColumn TItem="Product" TValue="decimal" Property="@(p => p.Price)"    Header="Price"    Sortable />
        <DataViewColumn TItem="Product" TValue="double"  Property="@(p => p.Rating)"   Header="Rating"   Sortable />
    </Fields>
</DataView>
