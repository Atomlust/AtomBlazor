@code {
    private void MoveTask(GanttMoveContext<PlanTask> move)
    {
        var item = move.Item;
        plan.Remove(item);

        // NewParentId is already worked out from where it landed: the target's own parent for a
        // drop between two rows, the target itself for a drop onto the middle of one.
        item.Parent = move.NewParentId;

        var target = plan.IndexOf(move.Target.Item);
        var at = move.Position == GanttDropPosition.Before ? target : target + 1;
        plan.Insert(Math.Clamp(at, 0, plan.Count), item);
    }

    private void Delete(PlanTask task)
    {
        // Everything under it goes too, or its children reappear at the top level: a parent
        // identifier naming a task that is not there is treated as no parent.
        var doomed = new HashSet<string> { task.Id };
        for (var pass = 0; pass < plan.Count; pass++)
        {
            foreach (var child in plan.Where(t => t.Parent is not null && doomed.Contains(t.Parent)))
            {
                doomed.Add(child.Id);
            }
        }

        plan = [.. plan.Where(t => !doomed.Contains(t.Id))];
    }
}

@* Adding is the same shape: put a new task in the collection and the chart draws it on the
   next render. A task cannot be dropped into its own branch; the chart refuses that itself
   rather than handing you a tree that is no longer a tree. *@
