@using System.ComponentModel.DataAnnotations
@using AtomBlazor.Primitives

<div class="flex items-center gap-2">
    <Checkbox Id="grid-reject-saves" @bind-Checked="rejectSaves" />
    <Label For="grid-reject-saves">Simulate a failed save</Label>
</div>
<DataGrid Items="validationRows" TData="Employee" ItemKey="@(r => r.Id)" EditMode="DataGridEditMode.Cell" EditItemFactory="Clone" OnRowCommit="SaveCellAsync" ShowPagination="false" AriaLabel="Editing validation and recovery">
    <Columns>
        <DataGridPropertyColumn TData="Employee" TProp="string" Property="@(r => r.Name)">
            <EditTemplate><InputGroupInput @bind-Value="context.Name" AriaLabel="Name" /></EditTemplate>
        </DataGridPropertyColumn>
        <DataGridPropertyColumn TData="Employee" TProp="int" Property="@(r => r.Hours)">
            <EditTemplate><NumericInput TValue="int" @bind-Value="context.Hours" AriaLabel="Hours" ShowButtons="false" /></EditTemplate>
        </DataGridPropertyColumn>
    </Columns>
</DataGrid>

@code {
    private readonly List<Employee> validationRows =
    [
        new() { Id = 1, Name = "Alex", Hours = 32 },
        new() { Id = 2, Name = "Sam", Hours = 24 },
        new() { Id = 3, Name = "Morgan", Hours = 40 },
        new() { Id = 4, Name = "Jamie", Hours = 16 }
    ];
    private bool rejectSaves;
    private async Task SaveCellAsync(DataGridRowCommitContext<Employee> context)
    {
        await Task.Delay(300);
        context.Cancel = rejectSaves;
    }
    private static Employee Clone(Employee source) => new() { Id = source.Id, Name = source.Name, Hours = source.Hours };
    private sealed class Employee
    {
        public int Id { get; set; }
        [Required] public string Name { get; set; } = "";
        [Range(0, 80)] public int Hours { get; set; }
    }
}
