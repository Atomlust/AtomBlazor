@using AtomBlazor.Primitives.Barcode

@* BarcodeEncoder returns the bars on their own - positions and widths in modules,
   heights as fractions of the bar area. Draw them however suits. *@
@code {
    private static void Dump()
    {
        var symbol = BarcodeEncoder.Encode("BB-42", BarcodeType.Code128);

        foreach (var bar in symbol.Bars)
        {
            // bar.Start and bar.Width are modules; bar.Top and bar.Bottom are 0 to 1.
            Console.WriteLine($"{bar.Start} {bar.Width} {bar.Top} {bar.Bottom}");
        }
    }
}
