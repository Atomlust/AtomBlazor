<Button Loading="true" LoadingText="Saving..." Variant="ButtonVariant.Outline">
    Save Changes
</Button>

<Button Loading="@_isLoading" OnClick="SimulateLoading">
    @(_isLoading ? "Submitting..." : "Submit")
</Button>

@code {
    private bool _isLoading;

    private async Task SimulateLoading(MouseEventArgs e)
    {
        _isLoading = true;
        StateHasChanged();
        await Task.Delay(2000);
        _isLoading = false;
    }
}
