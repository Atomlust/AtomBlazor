        <p>Custom navigation example - Step 1</p>
    </WizardStep>
    <WizardStep Title="Step 2">
        <p>Custom navigation example - Step 2</p>
    </WizardStep>
    <WizardStep Title="Step 3">
        <p>Custom navigation example - Step 3</p>
    </WizardStep>
</FormWizard>

    <Button Variant="ButtonVariant.Outline"
              Disabled="@(_step <= 0)"
              OnClick="() => _step = Math.Max(0, _step - 1)">
        Previous
    </Button>
    <Button Disabled="@(_step >= 2)"
              OnClick="() => _step = Math.Min(2, _step + 1)">
        Continue
    </Button>
</div>
