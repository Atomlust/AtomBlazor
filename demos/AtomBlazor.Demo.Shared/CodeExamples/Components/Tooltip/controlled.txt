<button @onclick="() => isOpen = !isOpen" class="px-3 py-2 border rounded text-sm">
    Toggle tooltip
</button>

<Tooltip @bind-Open="isOpen">
    <TooltipTrigger AsChild>
        <Button>@(isOpen ? "Hide" : "Show") tooltip</Button>
    </TooltipTrigger>
    <TooltipContent>Controlled by the parent component's state</TooltipContent>
</Tooltip>
