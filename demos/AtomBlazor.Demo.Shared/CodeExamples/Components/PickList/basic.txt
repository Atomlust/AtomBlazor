@* One binding, not two. Options holds everything; Values holds what has
   been picked. The available pane is whatever is left. *@
<PickList TValue="string"
            Options="@options"
            @bind-Values="picked" />

@code {
    private IEnumerable<string>? picked;

    private readonly SelectOption<string>[] options =
    [
        new("id", "ID"),
        new("name", "Name"),
        new("email", "Email"),
    ];
}
