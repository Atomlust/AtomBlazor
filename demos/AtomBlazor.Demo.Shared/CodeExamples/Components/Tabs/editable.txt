<Tabs @bind-Value="activeSheet">
    <TabsList Responsive Closable Renamable Reorderable Addable
                AddLabel="New sheet"
                OnAdd="AddSheet"
                OnClose="CloseSheet"
                OnRename="RenameSheet"
                OnMove="MoveSheet">
        @foreach (var sheet in sheets)
        {
            <TabsTrigger @key="sheet.Id"
                           Value="@sheet.Id"
                           Label="@sheet.Name"
                           Closable="@(sheets.Count > 1)" />
        }
    </TabsList>

    @foreach (var sheet in sheets)
    {
        <TabsContent @key="sheet.Id" Value="@sheet.Id">
            <p>The panel for @sheet.Name.</p>
        </TabsContent>
    }
</Tabs>

@code {
    private sealed record Sheet(string Id, string Name);

    private readonly List<Sheet> sheets =
    [
        new("summary", "Summary"),
        new("q1", "Q1"),
        new("q2", "Q2"),
    ];

    private string? activeSheet = "summary";
    private int nextSheet = 3;

    private void AddSheet()
    {
        nextSheet++;
        var id = $"sheet-{nextSheet}";
        sheets.Add(new Sheet(id, $"Sheet {nextSheet}"));
        activeSheet = id;
    }

    private void CloseSheet(string value)
    {
        var index = sheets.FindIndex(s => s.Id == value);
        if (index < 0 || sheets.Count == 1)
        {
            return;
        }

        sheets.RemoveAt(index);

        // The neighbour takes over when the tab that closed was the active one.
        if (activeSheet == value)
        {
            activeSheet = sheets[Math.Min(index, sheets.Count - 1)].Id;
        }
    }

    private void RenameSheet(TabRenameContext context)
    {
        var index = sheets.FindIndex(s => s.Id == context.Value);
        if (index < 0)
        {
            return;
        }

        // Set Cancel to refuse a rename and reopen the editor. Here, a name already in use.
        if (sheets.Any(s => s.Id != context.Value && s.Name == context.NewLabel))
        {
            context.Cancel = true;
            return;
        }

        sheets[index] = sheets[index] with { Name = context.NewLabel };
    }

    private void MoveSheet(TabMoveContext context)
    {
        var moved = sheets[context.OldIndex];
        sheets.RemoveAt(context.OldIndex);
        sheets.Insert(context.NewIndex, moved);
    }
}
