@* An event with an EventEnd on a later day is drawn as one bar across the days it
   covers. Overlapping runs stack, longest at the top, and a run crossing a week
   boundary becomes one bar per row with the join squared off.

   Nothing to switch on — this is just what a multi-day event looks like. *@
<EventCalendar Items="trips"
                 EventStart="e => e.Start"
                 EventEnd="e => e.End"
                 EventTitle="e => e.Title"
                 EventClass="e => e.Class"
                 MaxEventsPerDay="4" />

@code {
    private sealed record Trip(string Title, DateTime Start, DateTime? End, string Class);

    private static readonly DateTime monthStart =
        new(DateTime.Today.Year, DateTime.Today.Month, 1);

    private readonly List<Trip> trips = new()
    {
        // Two runs that overlap, so they take separate lanes.
        new("Onsite week", monthStart.AddDays(7), monthStart.AddDays(11), "bg-blue-500/15 text-blue-700 hover:bg-blue-500/25"),
        new("Design sprint", monthStart.AddDays(9), monthStart.AddDays(14), "bg-purple-500/15 text-purple-700 hover:bg-purple-500/25"),

        // Runs that never touch share one lane.
        new("Conference", monthStart.AddDays(2), monthStart.AddDays(4), "bg-emerald-500/15 text-emerald-700 hover:bg-emerald-500/25"),

        // A single-day event stays an ordinary chip.
        new("Retro", monthStart.AddDays(3), null, "bg-orange-500/15 text-orange-700 hover:bg-orange-500/25"),
    };
}
