@using AtomBlazor.Primitives.SignaturePad

@* Samples are in CSS pixels from the top left, so a narrower pad clips
   rather than scales. Replay onto a pad the same size. *@
@code {
    private SignaturePad? pad;
    private IReadOnlyList<SignatureStroke>? saved;

    private async Task SaveAsync() => saved = await pad!.GetStrokesAsync();

    private async Task RestoreAsync() => await pad!.SetStrokesAsync(saved);
}
