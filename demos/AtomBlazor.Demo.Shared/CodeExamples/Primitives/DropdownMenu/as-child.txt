@using AtomBlazor.Primitives.DropdownMenu

@* Without AsChild — trigger renders its own button *@
<DropdownMenu ItemClass="block px-2 py-1.5 cursor-pointer rounded hover:bg-accent focus:bg-accent focus:outline-none">
    <DropdownMenuTrigger class="...">Open Menu</DropdownMenuTrigger>
    <DropdownMenuContent class="...">
        <DropdownMenuItem>Item 1</DropdownMenuItem>
    </DropdownMenuContent>
</DropdownMenu>

@* With AsChild — your element becomes the trigger *@
<DropdownMenu ItemClass="...">
    <DropdownMenuTrigger AsChild>
        <Button Variant="ButtonVariant.Outline">Open Menu</Button>
    </DropdownMenuTrigger>
    ...
</DropdownMenu>
