<button @onclick="() => isControlledOpen = true">Open Sheet</button>
<button @onclick="() => isControlledOpen = false">Close from Outside</button>

<Sheet @bind-Open="isControlledOpen">
    <SheetPortal>
        <SheetOverlay class="fixed inset-0 bg-black/50 z-50" />
        <SheetContent class="fixed right-0 top-0 z-50 h-full w-3/4 sm:max-w-sm bg-background p-6 border-l">
            <SheetTitle>Controlled Sheet</SheetTitle>
            <SheetDescription>
                This sheet's state is controlled by the parent component.
            </SheetDescription>
            <p>Open state: <strong>@isControlledOpen</strong></p>
            <SheetClose class="px-3 py-2 border rounded text-sm">Close</SheetClose>
        </SheetContent>
    </SheetPortal>
</Sheet>

@code {
    private bool isControlledOpen = false;
}
